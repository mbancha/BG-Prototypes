export function applyAction(state:State,action:Action):string|null {const s=clone(state);try{if(action.type==='selection'){if(!['unit','card','tool'].includes(action.kind))return 'Invalid selection';}else if(action.type==='bid'){if(s.phase!=='bid'||!['stone','blue'].includes(action.zone)||(!Number.isSafeInteger(action.amount)||action.amount%2!==0)||action.amount<(s.round>1?1:0)||(s.round>1&&action.zone!==s.zones[s.side]))return 'Bids use whole PP; tie increments must be positive whole PP.';}else if(!legalActions(clone(s)).some(a=>same(a,action)))return 'That action is unavailable in the current decision.';execute(s,action);s.history.push(action);recordAction(s,action);Object.assign(state,s);return null;}catch(e){return 'Action rejected: '+String(e);}}
function execute(s:State,a:Action){const p=s.players[s.side];switch(a.type){case 'bid':{s.bids[s.side]={zone:a.zone,amount:a.amount};if(s.bids.every(Boolean)){s.reveal++;s.bidTotals=s.bidTotals.map((n,i)=>n+s.bids[i].amount);if(s.bids[0].zone===s.bids[1].zone&&s.bidTotals[0]===s.bidTotals[1]){s.round++;s.zones=s.bids.map(b=>b.zone);s.bids=[null,null];s.side=0;addLog(s,'Tied bids: submit positive secret increments.');break;}if(s.bids[0].zone!==s.bids[1].zone){s.zones=s.bids.map(b=>b.zone);s.first=Math.floor(rand(s)*2);}else{s.first=s.bidTotals[0]>s.bidTotals[1]?0:1;s.zones[s.first]=a.zone;s.zones[1-s.first]=a.zone==='stone'?'blue':'stone';s.players[1-s.first].pp+=s.bidTotals[s.first];s.players[1-s.first].bidBonus=s.bidTotals[s.first];}addLog(s,`Bids revealed: ${s.bidTotals.map(n=>n/2).join(' / ')} PP. ${s.players[s.first].name} goes first; budgets ${s.players.map(x=>x.pp/2).join(' / ')} PP.`);s.phase='prepare';s.side=0;}else s.side=1-s.side;break;}

 case 'move':{const u=unit(s,a.unit);s.ap--;if(u.hero){relocate(s,u,a.hex);}else{const models=activeUnits(s).filter(u=>u.hex);const ranges=Object.fromEntries(models.map(u=>[u.id,moveRange(s,u)]));u.hex=a.hex;const rest=models.filter(t=>t.id!==u.id).map(t=>t.id);if(rest.length)s.pending={kind:'groupMove',unit:rest[0],range:ranges[rest[0]],rest:rest.slice(1),ranges};}break;}
 case 'groupMove':{const models=activeUnits(s).filter(u=>u.hex);s.pending={kind:'groupMove',remaining:models.map(u=>u.id),ranges:Object.fromEntries(models.map(u=>[u.id,moveRange(s,u)]))};break;}
