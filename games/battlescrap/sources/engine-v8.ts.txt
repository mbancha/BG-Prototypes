function recordAction(s:State,a:Action){let detail='';const name=(id:string)=>id.endsWith('group')?'Recruit group':unit(s,id)?label(s,id):id;
 if(a.type==='selection')detail='selected '+a.kind+(a.kind==='unit'&&unit(s,a.value)?' '+name(a.value):a.kind==='tool'?' '+a.value:'');else if(a.type==='bid')detail='committed a secret bid';
 else if(['discard','discardObjective','reward'].includes(a.type))detail=a.type==='reward'?'chose '+a.source+' draw':'discarded a private card';
 else {detail=a.type.replace(/([A-Z])/g,' $1');if(a.unit)detail+=' '+name(a.unit);if(a.hero)detail+=' '+H[a.hero].name;if(a.target)detail+=' → '+name(a.target);if(a.hex)detail+=' → '+a.hex;if(a.suit)detail+=' '+a.suit;if(a.item)detail+=' '+a.item;if(a.card)detail+=' '+C[cardId(a.card)].name;if(a.type==='attackMode')detail=a.heavy?'chose heavy attack':'chose standard attack';if(a.type==='reroll')detail+=' die '+(a.die+1)+' for 1 PP';}
 addLog(s,`#${s.history.length} · ${detail} · ${s.ap} AP remaining`);
}
