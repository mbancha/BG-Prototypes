import {it,expect} from 'vitest';
import {newGame,actions,dispatch,bot,owner,playerView,nextBattle,replay,clone} from '../src/engine';
function go(s:any,a:any){expect(dispatch(s,a)).toBe(null);}
it('settles once, acquires only offered unfielded heroes with RP, and carries rosters into a fresh 20/21 PP battle',()=>{const s=newGame(8,{controllers:['bot','bot'],target:1});for(let n=0;n<2000&&s.phase!=='results';n++)go(s,bot(playerView(s,owner(s)),actions(s)));expect(s.phase).toBe('results');const winner=s.side,income=s.players.map(p=>p.rp),old=clone(s);expect(dispatch(s,{type:'resolve'})).not.toBe(null);expect(s).toEqual(old);const offer=actions(s).find(a=>a.type==='recruit');if(offer){const pp=s.players[s.side].pp;go(s,offer);expect(s.players[s.side].roster).toContain(offer.hero);expect(s.players[s.side].pp).toBe(pp);}for(let n=0;n<2;n++)go(s,{type:'nextResults'});expect(s.phase).toBe('complete');const n=nextBattle(s);expect(n.players[winner] .pp).toBe(40);expect(n.players[1-winner] .pp).toBe(42);expect(n.players.map(p=>p.rp)).toEqual(s.players.map(p=>p.rp));expect(n.players.map(p=>p.roster)).toEqual(s.players.map(p=>p.roster));expect(n.players.map(p=>p.hand)).toEqual([[],[]]);go(n,{type:'bid',zone:'blue',amount:2});expect(replay(n.seed,n.config,n.history)).toEqual(n);});
it('starting objectives wait until their owner actually activates',()=>{const s=newGame(1);for(let n=0;n<100&&s.phase!=='battle';n++)go(s,bot(playerView(s,owner(s)),actions(s)));expect(s.players.every(p=>p.objectives[0].awaiting)).toBe(true);const a=actions(s)[0];go(s,a);expect(s.players[s.side].objectives[0].awaiting).toBe(false);expect(s.players[1-s.side].objectives[0].awaiting).toBe(true);});
it('suit reassignment before the third hero cannot create duplicate suits',()=>{const s=newGame();go(s,{type:'bid',zone:'stone',amount:0});go(s,{type:'bid',zone:'blue',amount:0});go(s,{type:'hero',hero:'h05'});go(s,{type:'hero',hero:'h03'});go(s,{type:'suit',hero:'h05',suit:'Water'});go(s,{type:'hero',hero:'h04'});expect(new Set(s.players[0].selected.map(h=>s.players[0].suits[h])).size).toBe(3);});
