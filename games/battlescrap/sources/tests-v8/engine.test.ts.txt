import {describe,it,expect} from 'vitest';
import {newGame,actions,dispatch,clone,bot,playerView,owner,compare,replay,reachable,qualified} from '../src/engine';
import {H,heroes,cards,board,equipment} from '../src/content';
function step(s:any,a:any){const e=dispatch(s,a);expect(e,JSON.stringify(a)).toBe(null);}
function setup(seed=1){const s=newGame(seed,{controllers:['bot','bot']});for(let i=0;i<100&&s.phase!=='battle';i++){const a=bot(playerView(s,owner(s)),actions(s));if(!a)throw Error('No action '+s.phase);step(s,a);}expect(s.phase).toBe('battle');return s;}
describe('authored content',()=>{it('has ten exact heroes, twenty linked cards, six tactic effects in ten suit variants and affordable fixtures',()=>{expect(heroes.length).toBe(10);expect(cards.filter(c=>c.hero).length).toBe(20);expect(cards.filter(c=>!c.hero).length).toBe(10);const counts=[0,0];for(let p=0;p<2;p++){const f=heroes.slice(p*5,p*5+5);for(let a=0;a<3;a++)for(let b=a+1;b<4;b++)for(let c=b+1;c<5;c++)if(f[a].cost+f[b].cost+f[c].cost<=40)counts[p]++;}expect(counts.every(n=>n>0)).toBe(true);expect(H.h03.cost).toBe(10);expect(H.h09.cost).toBe(12);});});
describe('combat comparator',()=>{it.each([[[5],[5,5],true],[[6],[6,6],false],[[6,6,1],[6,6,5],true],[[6],[5,5,5],true]])('uses highest and only six counts: %j vs %j',(a:any,d:any,hit:any)=>expect(compare(a,d).hit).toBe(hit));it('requires two extra sixes for a critical',()=>{expect(compare([6,6],[5]).critical).toBe(true);expect(compare([6],[5]).critical).toBe(false);});});
describe('setup and currencies',()=>{it('keeps secret bids, handles whole PP ties and compensates only loser',()=>{const s=newGame();step(s,{type:'bid',zone:'stone',amount:6});expect(JSON.stringify(playerView(s,1))).not.toContain('"amount"');step(s,{type:'bid',zone:'stone',amount:6});expect(s.round).toBe(2);const before=clone(s);expect(dispatch(s,{type:'bid',zone:'stone',amount:0})).not.toBe(null);expect(s).toEqual(before);step(s,{type:'bid',zone:'stone',amount:2});step(s,{type:'bid',zone:'stone',amount:4});expect(s.first).toBe(1);expect(s.players.map(p=>p.pp)).toEqual([50,40]);});it('completes setup with unique queue tiles, exact decks and starting objectives',()=>{const s=setup();expect(new Set(s.queue).size).toBe(8);expect(s.units.filter(u=>u.hero).length).toBe(6);expect(s.units.filter(u=>!u.hero).length).toBe(6);for(const p of s.players){expect(p.hand.length+p.deck.length).toBe(24);expect(p.objectives.length).toBe(1);expect(p.pp).toBeGreaterThanOrEqual(0);}expect(s.objectiveDeck.length).toBe(8);});it('illegal commands never mutate state',()=>{const s=setup(),old=clone(s);expect(dispatch(s,{type:'move',unit:s.units[0].id,hex:'99,99'})).not.toBe(null);expect(s).toEqual(old);});});
describe('boundaries and reproducibility',()=>{it('omits opponent secrets, RNG and deck order',()=>{const s=setup(),v=playerView(s,0),t=clone(s);t.players[1].hand=['h10a:99'];t.rng=55;t.players[1].deck.reverse();expect(playerView(t,0).players[1].hand).toEqual([]);expect((v as any).rng).toBeUndefined();expect((v as any).seed).toBeUndefined();expect((v.players[0] as any).deck).toBeUndefined();});it('replays setup deterministically',()=>{const s=setup();expect(replay(s.seed,s.config,s.history)).toEqual(s);});it('continues a JSON save through identical commands',()=>{const s=setup(),t=JSON.parse(JSON.stringify(s));for(let i=0;i<35;i++){const a=bot(playerView(s,owner(s)),actions(s));step(s,a);step(t,a);}expect(t).toEqual(s);});});
describe('bounded complete bot games',()=>{it('finishes seeded matches while preserving board and card invariants',()=>{for(const seed of [1,2,3,4,5]){const s=setup(seed);let steps=0;for(;steps<1600&&!['results','complete'].includes(s.phase);steps++){const legal=actions(s);expect(legal.length,'No legal actions '+s.phase+' '+JSON.stringify(s.pending)).toBeGreaterThan(0);step(s,bot(playerView(s,owner(s)),legal));const positions=s.units.filter(u=>u.hex).map(u=>u.hex);expect(new Set(positions).size).toBe(positions.length);expect(s.queue.length).toBe(8);for(const p of s.players){expect(p.pp).toBeGreaterThanOrEqual(0);const all=[...p.hand,...p.deck,...p.discard];expect(all.length).toBe(24);expect(new Set(all).size).toBe(24);}}expect(s.phase,`seed ${seed} stalled after ${steps} / ${s.activation} activations`).toBe('results');expect(s.settled).toEqual([1]);}},60000);});
